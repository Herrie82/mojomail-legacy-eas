# Standalone build for testing without webOS SDK
cmake_minimum_required(VERSION 3.10)

project(mojomail-eas CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_compile_options(-DMOJ_LINUX -O2 -g -Wall)

include_directories(inc)
include_directories(${CMAKE_SOURCE_DIR}/../common/inc)

# Find packages
find_package(PkgConfig REQUIRED)
find_package(Boost REQUIRED COMPONENTS regex)
find_package(CURL REQUIRED)

pkg_check_modules(GLIB2 REQUIRED glib-2.0)

include_directories(
    ${Boost_INCLUDE_DIRS}
    ${CURL_INCLUDE_DIRS}
    ${GLIB2_INCLUDE_DIRS}
)

# Source files
set(WBXML_SOURCES
    src/wbxml/WbxmlEncoder.cpp
    src/wbxml/WbxmlParser.cpp
    src/wbxml/EasTagCode.cpp
)

set(COMMAND_SOURCES
    src/commands/EasCommand.cpp
    src/commands/FolderSyncCommand.cpp
    src/commands/OptionsCommand.cpp
    src/commands/ProvisionCommand.cpp
    src/commands/PingCommand.cpp
    src/commands/SyncCommand.cpp
)

set(DATA_SOURCES
    src/data/EasAccount.cpp
    src/data/EasFolder.cpp
    src/data/EasEmailAdapter.cpp
    src/data/EasContactAdapter.cpp
    src/data/EasEventAdapter.cpp
    src/data/EasTaskAdapter.cpp
)

set(CORE_SOURCES
    src/EasClient.cpp
    src/EasBusDispatcher.cpp
    src/EasConfig.cpp
    src/EasServiceApp.cpp
)

# For standalone build, just compile to check syntax
add_library(mojomail-eas-lib STATIC
    ${WBXML_SOURCES}
    ${COMMAND_SOURCES}
    ${DATA_SOURCES}
    ${CORE_SOURCES}
)

target_link_libraries(mojomail-eas-lib
    ${GLIB2_LIBRARIES}
    ${CURL_LIBRARIES}
    ${Boost_LIBRARIES}
)
